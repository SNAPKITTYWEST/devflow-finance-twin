dcl-proc PostTransaction;
  dcl-pi *n;
    pTransId packed(15:0) const;
  end-pi;

  transId = pTransId;
  exec sql set option commit = *chg, usingsqlca = *yes;

  // Load transaction
  exec sql
    select IDEMPOTENCY_KEY, CORRELATION_ID, STATUS
      into :idemKey, :corrId, :*inlr
      from TRANSACTIONS
     where TRANSACTION_ID = :transId
       for update of STATUS;

  sqlState = sqlstate;
  if sqlState <> '00000';
     callp LogError('POSTTRAN':'TXN_LOAD':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  // Idempotency: if already POSTED, return without side effects
  if %trim(*inlr) = 'POSTED';
     // No-op but log audit
     callp LogAudit('SYSTEM':'TransactionAlreadyPosted':'TRANSACTION':transId:corrId);
     commit;
     *inlr = *on;
     return;
  endif;

  // Create journal header
  exec sql
    insert into JOURNALS (
      LEDGER_ID, TRANSACTION_ID, EFFECTIVE_DATE, POSTING_DATE,
      STATUS, SOURCE, IDEMPOTENCY_KEY, CORRELATION_ID, CREATED_AT
    )
    values (
      (select LEDGER_ID from ACCOUNTS a
         join TRANSACTIONS t on a.ACCOUNT_ID = t.ACCOUNT_ID
        where t.TRANSACTION_ID = :transId),
      :transId,
      current_date,
      current_timestamp,
      'PENDING',
      'C#-API',
      :idemKey,
      :corrId,
      current_timestamp
    );

  if sqlcode <> 0;
     sqlState = sqlstate;
     callp LogError('POSTTRAN':'JOURNAL_INSERT':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  exec sql
    select JOURNAL_ID into :journalId
      from JOURNALS
     where TRANSACTION_ID = :transId
       and IDEMPOTENCY_KEY = :idemKey;

  if sqlcode <> 0;
     sqlState = sqlstate;
     callp LogError('POSTTRAN':'JOURNAL_SELECT':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  // Build debit/credit lines (simplified: single account + offset account)
  dcl-s accountId packed(15:0);
  dcl-s offsetAccountId packed(15:0);
  dcl-s amount decimal(18:4);
  dcl-s currency char(3);
  dcl-s direction char(1);

  exec sql
    select ACCOUNT_ID, AMOUNT, CURRENCY_CODE, DIRECTION
      into :accountId, :amount, :currency, :direction
      from TRANSACTIONS
     where TRANSACTION_ID = :transId;

  if sqlcode <> 0;
     sqlState = sqlstate;
     callp LogError('POSTTRAN':'TXN_SELECT':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  // Determine offset account (e.g., settlement or cash account)
  exec sql
    select OFFSET_ACCOUNT_ID
      into :offsetAccountId
      from ACCOUNT_TYPES at
      join ACCOUNTS a on a.ACCOUNT_TYPE_ID = at.ACCOUNT_TYPE_ID
     where a.ACCOUNT_ID = :accountId;

  if sqlcode <> 0 or offsetAccountId = 0;
     sqlState = sqlstate;
     callp LogError('POSTTRAN':'OFFSET_ACCOUNT':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  // Insert journal lines: one debit, one credit
  if direction = 'D';
     // Debit primary, credit offset
     exec sql
       insert into JOURNAL_LINES (
         JOURNAL_ID, ACCOUNT_ID, AMOUNT, CURRENCY_CODE, DIRECTION
       ) values (
         :journalId, :accountId, :amount, :currency, 'D'
       );
     exec sql
       insert into JOURNAL_LINES (
         JOURNAL_ID, ACCOUNT_ID, AMOUNT, CURRENCY_CODE, DIRECTION
       ) values (
         :journalId, :offsetAccountId, :amount, :currency, 'C'
       );
  else;
     // Credit primary, debit offset
     exec sql
       insert into JOURNAL_LINES (
         JOURNAL_ID, ACCOUNT_ID, AMOUNT, CURRENCY_CODE, DIRECTION
       ) values (
         :journalId, :accountId, :amount, :currency, 'C'
       );
     exec sql
       insert into JOURNAL_LINES (
         JOURNAL_ID, ACCOUNT_ID, AMOUNT, CURRENCY_CODE, DIRECTION
       ) values (
         :journalId, :offsetAccountId, :amount, :currency, 'D'
       );
  endif;

  if sqlcode <> 0;
     sqlState = sqlstate;
     callp LogError('POSTTRAN':'JLINE_INSERT':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  // Validate double-entry invariant: SUM(D) = SUM(C)
  exec sql
    select coalesce(sum(case when DIRECTION = 'D' then AMOUNT else 0 end),0),
           coalesce(sum(case when DIRECTION = 'C' then AMOUNT else 0 end),0)
      into :debitTotal, :creditTotal
      from JOURNAL_LINES
     where JOURNAL_ID = :journalId;

  if debitTotal <> creditTotal;
     callp LogError('POSTTRAN':'IMBALANCED_JOURNAL':corrId:'00000');
     rollback;
     *inlr = *on;
     return;
  endif;

  // Update balances atomically
  callp UpdateBalance(accountId:direction:amount:corrId);
  callp UpdateBalance(offsetAccountId:
                      (case direction; when 'D' => 'C'; other; 'D'; endcase):
                      amount:corrId);

  if %error;
     // UpdateBalance must set error state via SQLCA
     sqlState = sqlstate;
     callp LogError('POSTTRAN':'BALANCE_UPDATE':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  // Mark journal and transaction as POSTED
  exec sql
    update JOURNALS
       set STATUS = 'POSTED'
     where JOURNAL_ID = :journalId;

  exec sql
    update TRANSACTIONS
       set STATUS = 'POSTED',
           UPDATED_AT = current_timestamp
     where TRANSACTION_ID = :transId;

  if sqlcode <> 0;
     sqlState = sqlstate;
     callp LogError('POSTTRAN':'STATUS_UPDATE':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  // Audit event
  callp LogAudit('SYSTEM':'TransactionPosted':'TRANSACTION':transId:corrId);

  commit;
  *inlr = *on;
end-proc;
dcl-proc UpdateBalance;
  dcl-pi *n;
    pAccountId packed(15:0) const;
    pDirection char(1) const;
    pAmount    decimal(18:4) const;
    pCorrId    char(128) const;
  end-pi;

  dcl-s sqlState char(5);
  dcl-s ledgerBal decimal(18:4);
  dcl-s availBal  decimal(18:4);

  exec sql
    select LEDGER_BALANCE, AVAILABLE_BAL
      into :ledgerBal, :availBal
      from BALANCES
     where ACCOUNT_ID = :pAccountId
       for update of LEDGER_BALANCE, AVAILABLE_BAL;

  sqlState = sqlstate;
  if sqlState <> '00000';
     callp LogError('UPDBAL':'BAL_SELECT':pCorrId:sqlState);
     rollback;
     return;
  endif;

  select;
  when pDirection = 'D';
     ledgerBal += pAmount;
     availBal  += pAmount;
  when pDirection = 'C';
     ledgerBal -= pAmount;
     availBal  -= pAmount;
  other;
     callp LogError('UPDBAL':'INVALID_DIRECTION':pCorrId:'00000');
     rollback;
     return;
  endsl;

  exec sql
    update BALANCES
       set LEDGER_BALANCE = :ledgerBal,
           AVAILABLE_BAL  = :availBal,
           UPDATED_AT     = current_timestamp
     where ACCOUNT_ID = :pAccountId;

  sqlState = sqlstate;
  if sqlState <> '00000';
     callp LogError('UPDBAL':'BAL_UPDATE':pCorrId:sqlState);
     rollback;
     return;
  endif;
end-proc;

end-proc;

public sealed class LedgerService : ILedgerService
{
    private readonly IRpgInvoker _rpgInvoker; // bridge to IBM i programs
    private readonly ILogger<LedgerService> _logger;

    public LedgerService(IRpgInvoker rpgInvoker, ILogger<LedgerService> logger)
    {
        _rpgInvoker = rpgInvoker;
        _logger = logger;
    }

    public async Task<TransactionPostedResponse> PostTransactionAsync(
        PostTransactionRequest request,
        CancellationToken ct)
    {
        // Idempotency key enforced by RPG/DB2
        var rpgRequest = new RpgPostTransactionRequest
        {
            TransactionId = request.TransactionId,
            CorrelationId = request.CorrelationId,
            IdempotencyKey = request.IdempotencyKey
        };

        var result = await _rpgInvoker.InvokeAsync("POSTTRAN", rpgRequest, ct);

        if (!result.Success)
        {
            _logger.LogWarning("Posting failed {@Result}", result);
            return new TransactionPostedResponse
            {
                Success = false,
                Status = "FAILED",
                ErrorCode = result.ErrorCode,
                CorrelationId = request.CorrelationId
            };
        }

        return new TransactionPostedResponse
        {
            Success = true,
            Status = "POSTED",
            CorrelationId = request.CorrelationId
        };
    }

    public Task<AccountBalanceResponse> GetAccountBalanceAsync(long accountId, CancellationToken ct)
    {
        // Read-only query to BALANCES table (no ledger duplication)
        throw new NotImplementedException();
    }

public class IdempotencyMiddleware : IMiddleware
{
    private readonly IIdempotencyStore _store;
    private readonly ILogger<IdempotencyMiddleware> _logger;

    public IdempotencyMiddleware(IIdempotencyStore store, ILogger<IdempotencyMiddleware> logger)
    {
        _store = store;
        _logger = logger;
    }

    public async Task InvokeAsync(HttpContext context, RequestDelegate next)
    {
        var key = context.Request.Headers["Idempotency-Key"].FirstOrDefault();
        var correlationId = context.TraceIdentifier;

        if (string.IsNullOrWhiteSpace(key))
        {
            await next(context);
            return;
        }

        var existing = await _store.GetAsync(key, context.Request.Path, context.Request.Method);

        if (existing is not null)
        {
            _logger.LogInformation("Idempotent replay {Key} {CorrelationId}", key, correlationId);
            context.Response.StatusCode = existing.StatusCode;
            await context.Response.WriteAsync(existing.Body);
            return;
        }

        // Capture response
        var originalBody = context.Response.Body;
        using var buffer = new MemoryStream();
        context.Response.Body = buffer;

        await next(context);

        buffer.Seek(0, SeekOrigin.Begin);
        var bodyText = await new StreamReader(buffer).ReadToEndAsync();
        buffer.Seek(0, SeekOrigin.Begin);
        await buffer.CopyToAsync(originalBody);
        context.Response.Body = originalBody;

        await _store.SaveAsync(new IdempotencyRecord
        {
            Key = key,
            Path = context.Request.Path,
            Method = context.Request.Method,
            StatusCode = context.Response.StatusCode,
            Body = bodyText,
            CreatedAt = DateTimeOffset.UtcNow
        });
    }

dcl-proc PostTransaction;
  dcl-pi *n;
    pTransId packed(15:0) const;
  end-pi;

  transId = pTransId;

  exec sql set option commit = *chg, usingsqlca = *yes;

  // Load transaction row FOR UPDATE
  exec sql
    select ACCOUNT_ID, AMOUNT, CURRENCY_CODE, DIRECTION,
           IDEMPOTENCY_KEY, CORRELATION_ID, STATUS
      into :accountId, :amount, :currency, :direction,
           :idemKey, :corrId, :status
      from TRANSACTIONS
     where TRANSACTION_ID = :transId
       for update of STATUS;

  sqlState = sqlstate;
  if sqlState <> '00000';
     LogError('POSTTRAN':'TXN_LOAD':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  // Idempotency: already POSTED? no-op
  if %trim(status) = 'POSTED';
     commit;
     *inlr = *on;
     return;
  endif;

  // Determine offset account (hard-coded for demo)
  offsetAccountId = 1; // e.g., settlement account

  // Create journal header
  exec sql
    insert into JOURNALS (
      TRANSACTION_ID, EFFECTIVE_DATE, POSTING_DATE,
      STATUS, IDEMPOTENCY_KEY, CORRELATION_ID, CREATED_AT
    ) values (
      :transId,
      current_date,
      current_timestamp,
      'PENDING',
      :idemKey,
      :corrId,
      current_timestamp
    );

  if sqlcode <> 0;
     sqlState = sqlstate;
     LogError('POSTTRAN':'JOURNAL_INSERT':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  exec sql
    select JOURNAL_ID into :journalId
      from JOURNALS
     where TRANSACTION_ID = :transId
       and IDEMPOTENCY_KEY = :idemKey;

  if sqlcode <> 0;
     sqlState = sqlstate;
     LogError('POSTTRAN':'JOURNAL_SELECT':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  // Insert debit/credit lines
  if direction = 'D';
     exec sql
       insert into JOURNAL_LINES (
         JOURNAL_ID, ACCOUNT_ID, AMOUNT, CURRENCY_CODE, DIRECTION
       ) values (
         :journalId, :accountId, :amount, :currency, 'D'
       );
     exec sql
       insert into JOURNAL_LINES (
         JOURNAL_ID, ACCOUNT_ID, AMOUNT, CURRENCY_CODE, DIRECTION
       ) values (
         :journalId, :offsetAccountId, :amount, :currency, 'C'
       );
  else;

       insert into JOURNAL_LINES (
         JOURNAL_ID, ACCOUNT_ID, AMOUNT, CURRENCY_CODE, DIRECTION
       ) values (
         :journalId, :accountId, :amount, :currency, 'C'
       );
     exec sql
       insert into JOURNAL_LINES (
         JOURNAL_ID, ACCOUNT_ID, AMOUNT, CURRENCY_CODE, DIRECTION
       ) values (
         :journalId, :offsetAccountId, :amount, :currency, 'D'
       );
  endif;

  if sqlcode <> 0;
     sqlState = sqlstate;
     LogError('POSTTRAN':'JLINE_INSERT':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  // Double-entry invariant
  exec sql
    select coalesce(sum(case when DIRECTION = 'D' then AMOUNT else 0 end),0),
           coalesce(sum(case when DIRECTION = 'C' then AMOUNT else 0 end),0)
      into :debitTotal, :creditTotal
      from JOURNAL_LINES
     where JOURNAL_ID = :journalId;

  if debitTotal <> creditTotal;
     LogError('POSTTRAN':'IMBALANCED_JOURNAL':corrId:'00000');
     rollback;
     *inlr = *on;
     return;
  endif;

  // Update balances
  UpdateBalance(accountId:direction:amount:corrId);
  if direction = 'D';
     UpdateBalance(offsetAccountId:'C':amount:corrId);
  else;
     UpdateBalance(offsetAccountId:'D':amount:corrId);
  endif;

  // Mark journal and transaction as POSTED
  exec sql
    update JOURNALS
       set STATUS = 'POSTED'
     where JOURNAL_ID = :journalId;

    update TRANSACTIONS
       set STATUS = 'POSTED',
           UPDATED_AT = current_timestamp
     where TRANSACTION_ID = :transId;

  if sqlcode <> 0;
     sqlState = sqlstate;
     LogError('POSTTRAN':'STATUS_UPDATE':corrId:sqlState);
     rollback;
     *inlr = *on;
     return;
  endif;

  commit;
  *inlr = *on;
end-proc;

     rollback;

  endif;

// Ledger gateway: C# -> DB2 row + COBOL FSL + RPG
public interface ILedgerGateway
{
    Task<(bool Success, long TransactionId)> InitiateAndPostAsync(
        PostTransactionRequest request,
        string correlationId,
        CancellationToken ct);
}

public sealed class LedgerGateway : ILedgerGateway
{
    private readonly IDbConnection _db;        // DB2 connection
    private readonly ICobolInvoker _cobol;     // bridge to COBOL FSL

    public LedgerGateway(IDbConnection db, ICobolInvoker cobol)
    {
        _db = db;
        _cobol = cobol;
    }

    public async Task<(bool Success, long TransactionId)> InitiateAndPostAsync(
        PostTransactionRequest request,
        string correlationId,
        CancellationToken ct)
    {
        using var tx = _db.BeginTransaction();

        // Create TRANSACTIONS row (INITIATED)
        var cmd = _db.CreateCommand();
        cmd.Transaction = tx;
        cmd.CommandText = @"
            INSERT INTO TRANSACTIONS (
                ACCOUNT_ID, AMOUNT, CURRENCY_CODE, DIRECTION,
                STATUS, IDEMPOTENCY_KEY, CORRELATION_ID,
                CREATED_AT, UPDATED_AT
            ) VALUES (
                @acct, @amt, @cur, @dir,
                'INITIATED', @idem, @corr,
                CURRENT_TIMESTAMP, CURRENT_TIMESTAMP
            );
            SELECT IDENTITY_VAL_LOCAL() FROM SYSIBM.SYSDUMMY1";
        cmd.Parameters.Add(newParameter(cmd, "@acct", request.AccountId));
        cmd.Parameters.Add(newParameter(cmd, "@amt", request.Amount));
        cmd.Parameters.Add(newParameter(cmd, "@cur", request.CurrencyCode));
        cmd.Parameters.Add(newParameter(cmd, "@dir", request.Direction));
        cmd.Parameters.Add(newParameter(cmd, "@idem", request.IdempotencyKey));
        cmd.Parameters.Add(newParameter(cmd, "@corr", correlationId));

        var txnId = Convert.ToInt64(await cmd.ExecuteScalarAsync(ct));

        tx.Commit();

        // Call COBOL FSL supervisor, which will call RPG POSTTRAN
        var cobolRequest = new CobolTransactionState
        {
            TransactionId = txnId
        };

        var result = await _cobol.InvokeAsync("TXN-FSL-SUPERVISOR", cobolRequest, ct);

        return (result.Success, txnId);
    }

    private static IDbDataParameter newParameter(IDbCommand cmd, string name, object value)
    {
        var p = cmd.CreateParameter();
        p.ParameterName = name;
        p.Value = value;
        return p;
    }
}

      into :accountId, :amount, :currency, :direction, :sqlState
      from TRANSACTIONS

       DATA DIVISION.
       WORKING-STORAGE SECTION.

       01  ACH-INSTRUCTION.
           05  ACH-ID              PIC 9(15) COMP-3.
           05  PAYMENT-ID          PIC 9(15) COMP-3.
           05  TX-ID               PIC 9(15) COMP-3.
           05  AMOUNT              PIC 9(15)V9(4) COMP-3.
           05  CURRENCY            PIC X(3).
           05  STATUS              PIC X(32).
           05  RAIL-CODE           PIC X(8).
           05  NACHA-TRACE         PIC X(32).
           05  CORRELATION-ID      PIC X(128).

       01  FLAGS.
           05  CAN-SUBMIT          PIC 9(1) COMP-3 VALUE 0.
           05  CAN-SETTLE          PIC 9(1) COMP-3 VALUE 0.
           05  CAN-RETURN          PIC 9(1) COMP-3 VALUE 0.

       01  SQLSTATE-FIELD          PIC X(5).

       PROCEDURE DIVISION.

       MAIN-LOGIC.
           PERFORM LOAD-ACH
           PERFORM EVALUATE-ACH-STATE
           IF CAN-SUBMIT = 1
               PERFORM SUBMIT-ACH
           END-IF
           IF CAN-SETTLE = 1
               PERFORM SETTLE-ACH
           END-IF
           IF CAN-RETURN = 1
               PERFORM RETURN-ACH
           END-IF
           GOBACK.

       LOAD-ACH.
           EXEC SQL
                SELECT ACH_BATCH_ID,
                       PAYMENT_ID,
                       TRANSACTION_ID,
                       AMOUNT,
                       CURRENCY_CODE,
                       STATUS,
                       'ACH',
                       TRACE_NUMBER,
                       CORRELATION_ID
                  INTO :ACH-ID,
                       :PAYMENT-ID,
                       :TX-ID,
                       :AMOUNT,
                       :CURRENCY,
                       :STATUS,
                       :RAIL-CODE,
                       :NACHA-TRACE,
                       :CORRELATION-ID
                  FROM ACH_ENTRIES
                 WHERE ACH_ENTRY_ID = :ACH-ID
           END-EXEC
           MOVE SQLSTATE TO SQLSTATE-FIELD
           IF SQLSTATE-FIELD NOT = "00000"
               PERFORM LOG-ERROR
           END-IF
           .

       EVALUATE-ACH-STATE.
           EVALUATE STATUS
               WHEN "CREATED"
                    MOVE 1 TO CAN-SUBMIT
               WHEN "SUBMITTED"
                    MOVE 1 TO CAN-SETTLE
               WHEN "SETTLED"
                    MOVE 0 TO CAN-SETTLE
               WHEN "RETURN_PENDING"
                    MOVE 1 TO CAN-RETURN
               WHEN OTHER
                    MOVE 0 TO CAN-SUBMIT
                    MOVE 0 TO CAN-SETTLE
                    MOVE 0 TO CAN-RETURN
           END-EVALUATE
           .

       SUBMIT-ACH.
           * here you would generate NACHA file, mark status SUBMITTED
           EXEC SQL
                UPDATE ACH_ENTRIES
                   SET STATUS = 'SUBMITTED'
                 WHERE ACH_ENTRY_ID = :ACH-ID
           END-EXEC
           .

       SETTLE-ACH.
           * settlement triggers ledger posting via RPG
           CALL "POSTTRAN" USING TX-ID
           IF RETURN-CODE NOT = 0
               PERFORM LOG-ERROR
           ELSE
               EXEC SQL
                    UPDATE ACH_ENTRIES
                       SET STATUS = 'SETTLED'
                     WHERE ACH_ENTRY_ID = :ACH-ID
               END-EXEC
           END-IF
           .

       RETURN-ACH.
           * return triggers reversal via RPG
           CALL "REVTRAN" USING TX-ID
           IF RETURN-CODE NOT = 0
               PERFORM LOG-ERROR
           ELSE
               EXEC SQL
                    UPDATE ACH_ENTRIES
                       SET STATUS = 'RETURNED'
                     WHERE ACH_ENTRY_ID = :ACH-ID
               END-EXEC
           END-IF
           .

       LOG-ERROR.
           EXEC SQL
                INSERT INTO ERROR_QUEUE (
                    SOURCE, CORRELATION_ID, ERROR_CODE, MESSAGE, CREATED_AT
                ) VALUES (
                    'ACH-FSL',
                    :CORRELATION-ID,
                    'ACH_FSL_ERROR',
                    'ACH STATE OR LEDGER FAILURE',
                    CURRENT_TIMESTAMP
                )
           END-EXEC

[ApiController]
[Route("payments")]
public class PaymentsController : ControllerBase
{
    private readonly ILedgerGateway _ledger;
    private readonly IAchRailService _achRail;

    public PaymentsController(ILedgerGateway ledger, IAchRailService achRail)
    {
        _ledger = ledger;
        _achRail = achRail;
    }

    [HttpPost]
    public async Task<ActionResult<CreatePaymentResponse>> CreatePayment(
        [FromBody] CreatePaymentRequest request,
        CancellationToken ct)
    {
        var correlationId = HttpContext.TraceIdentifier;

        // 1. Create transaction + post via ledger gateway
        var postReq = new PostTransactionRequest(
            request.DebitAccountId,
            request.Amount,
            request.CurrencyCode,
            "D",
            request.IdempotencyKey);

        var postResult = await _ledger.InitiateAndPostAsync(postReq, correlationId, ct);

        if (!postResult.Success)
            return StatusCode(500, new CreatePaymentResponse(
                0, "FAILED", correlationId));

        // 2. Create ACH entry (rail metadata only)
        var achId = await _achRail.CreateAchEntryAsync(
            new AchEntryRequest
            {
                PaymentId = request.PaymentId,
                TransactionId = postResult.TransactionId,
                Amount = request.Amount,
                CurrencyCode = request.CurrencyCode,
                CorrelationId = correlationId
            },
            ct);

        return Ok(new CreatePaymentResponse(
            achId,
            "SUBMITTED",
            correlationId));
    }

    [HttpPost("{id}/reverse")]
    public async Task<ActionResult<ReversePaymentResponse>> ReversePayment(
        long id,
        [FromBody] ReversePaymentRequest request,
        CancellationToken ct)
    {
        var correlationId = HttpContext.TraceIdentifier;

        // Lookup original transaction for this payment
        var txId = await _achRail.GetTransactionIdForPaymentAsync(id, ct);

        // Call RPG reversal via gateway
        var result = await _ledger.ReverseAsync(txId, request.IdempotencyKey, correlationId, ct);

        if (!result.Success)
            return StatusCode(500, new ReversePaymentResponse(
                id, "FAILED", correlationId));

        return Ok(new ReversePaymentResponse(
            id, "REVERSED", correlationId));
    }
}

          05 CAN-SUBMIT          PIC 9(1) COMP-3 VALUE 0.
          05 CAN-SETTLE          PIC 9(1) COMP-3 VALUE 0.
          05 CAN-RETURN          PIC 9(1) COMP-3 VALUE 0.

       PROCEDURE DIVISION.

       MAIN.
           PERFORM LOAD-ACH
           PERFORM EVAL-STATE
           IF CAN-SUBMIT = 1
              PERFORM SUBMIT
           END-IF
           IF CAN-SETTLE = 1
              PERFORM SETTLE
           END-IF
           IF CAN-RETURN = 1
              PERFORM RETURN
           END-IF
           GOBACK.

       LOAD-ACH.
           EXEC SQL
                SELECT ACH_ENTRY_ID,
                       PAYMENT_ID,
                       TRANSACTION_ID,
                       AMOUNT,
                       CURRENCY_CODE,
                       STATUS,
                       CORRELATION_ID
                  INTO :ACH-ID,
                       :PAYMENT-ID,
                       :TX-ID,
                       :AMOUNT,
                       :CURRENCY,
                       :STATUS,
                       :CORR-ID
                  FROM ACH_ENTRIES
                 WHERE ACH_ENTRY_ID = :ACH-ID
           END-EXEC.

       EVAL-STATE.
           EVALUATE STATUS
              WHEN "CREATED"
                 MOVE 1 TO CAN-SUBMIT
              WHEN "SUBMITTED"
                 MOVE 1 TO CAN-SETTLE
              WHEN "RETURN_PENDING"
                 MOVE 1 TO CAN-RETURN
              WHEN OTHER
                 MOVE 0 TO CAN-SUBMIT
                 MOVE 0 TO CAN-SETTLE
                 MOVE 0 TO CAN-RETURN
           END-EVALUATE.

       SUBMIT.
           EXEC SQL
                UPDATE ACH_ENTRIES
                   SET STATUS='SUBMITTED'
                 WHERE ACH_ENTRY_ID=:ACH-ID
           END-EXEC.

       SETTLE.
           CALL "POSTTRAN" USING TX-ID
           IF RETURN-CODE = 0
              EXEC SQL
                   UPDATE ACH_ENTRIES
                      SET STATUS='SETTLED'
                    WHERE ACH_ENTRY_ID=:ACH-ID
              END-EXEC
           ELSE
              PERFORM LOG-ERR
           END-IF.

       RETURN.
           CALL "REVTRAN" USING TX-ID
           IF RETURN-CODE = 0
              EXEC SQL
                   UPDATE ACH_ENTRIES
                      SET STATUS='RETURNED'
                    WHERE ACH_ENTRY_ID=:ACH-ID
              END-EXEC
           ELSE
              PERFORM LOG-ERR
           END-IF.

       LOG-ERR.
           EXEC SQL
                INSERT INTO ERROR_QUEUE (
                    SOURCE, CORRELATION_ID, ERROR_CODE, MESSAGE, CREATED_AT
                ) VALUES (
                    'ACH-FSL',
                    :CORR-ID,
                    'ACH_FSL_ERROR',
                    'STATE FAILURE',
                    CURRENT_TIMESTAMP
                )
           END-EXEC.

public class PaymentsController : ControllerBase
{
    private readonly ILedgerGateway _ledger;
    private readonly IAchRail _ach;

    public PaymentsController(ILedgerGateway ledger, IAchRail ach)
    {
        _ledger = ledger;
        _ach = ach;
    }

    [HttpPost]
    public async Task<ActionResult<CreatePaymentResponse>> CreatePayment(
        CreatePaymentRequest req,
        CancellationToken ct)
    {
        var corr = HttpContext.TraceIdentifier;

        var postReq = new PostTransactionRequest(
            req.DebitAccountId,
            req.Amount,
            req.CurrencyCode,
            "D",
            req.IdempotencyKey);

        var post = await _ledger.InitiateAndPostAsync(postReq, corr, ct);

        if (!post.Success)
            return StatusCode(500, new CreatePaymentResponse(0, "FAILED", corr));

        var achId = await _ach.CreateEntryAsync(
            new AchEntryRequest
            {
                PaymentId = req.PaymentId,
                TransactionId = post.TransactionId,
                Amount = req.Amount,
                CurrencyCode = req.CurrencyCode,
                CorrelationId = corr
            },
            ct);

        return Ok(new CreatePaymentResponse(achId, "SUBMITTED", corr));
    }

    [HttpPost("{id}/reverse")]
    public async Task<ActionResult<ReversePaymentResponse>> Reverse(long id,
        ReversePaymentRequest req,
        CancellationToken ct)
    {
        var corr = HttpContext.TraceIdentifier;

        var txId = await _ach.GetTransactionIdAsync(id, ct);

        var rev = await _ledger.ReverseAsync(txId, req.IdempotencyKey, corr, ct);

        if (!rev.Success)
            return StatusCode(500, new ReversePaymentResponse(id, "FAILED", corr));

        return Ok(new ReversePaymentResponse(id, "REVERSED", corr));
    }

// Binary record layout for transaction command
// Matches COBOL + RPG expectations exactly.
[StructLayout(LayoutKind.Sequential, Pack = 1)]
public struct LedgerCommandRecord
{
    public long TransactionId;              // PIC 9(15) COMP-3 mapped as 8-byte
    public long AccountId;                  // PIC 9(15) COMP-3
    public decimal Amount;                  // DECIMAL(18,4) encoded as 16 bytes
    [MarshalAs(UnmanagedType.ByValTStr, SizeConst = 3)]
    public string Currency;                 // CHAR(3)
    [MarshalAs(UnmanagedType.ByValTStr, SizeConst = 1)]
    public string Direction;                // 'D' or 'C'
    [MarshalAs(UnmanagedType.ByValTStr, SizeConst = 128)]
    public string IdempotencyKey;           // CHAR(128)
    [MarshalAs(UnmanagedType.ByValTStr, SizeConst = 128)]
    public string CorrelationId;            // CHAR(128)
}
public interface ICobolGateway
{
    Task<bool> InvokeLedgerCommandAsync(LedgerCommandRecord record,
                                        string programName,
                                        CancellationToken ct);
}

public sealed class CobolGateway : ICobolGateway
{
    private readonly IIbmProgramInvoker _invoker; // abstraction over IBM i program call

    public CobolGateway(IIbmProgramInvoker invoker)
    {
        _invoker = invoker;
    }

    public async Task<bool> InvokeLedgerCommandAsync(LedgerCommandRecord record,
                                                     string programName,
                                                     CancellationToken ct)
    {
        var size = Marshal.SizeOf<LedgerCommandRecord>();
        var buffer = new byte[size];
        var ptr = Marshal.AllocHGlobal(size);
        try
        {
            Marshal.StructureToPtr(record, ptr, false);
            Marshal.Copy(ptr, buffer, 0, size);

            // Binary call: single parameter, fixed-length buffer
            var result = await _invoker.CallAsync(programName, buffer, ct);
            return result.Success;
        }
        finally
        {
            Marshal.FreeHGlobal(ptr);
        }
    }       IDENTIFICATION DIVISION.
       PROGRAM-ID. LEDGER-GATEWAY.

       DATA DIVISION.
       WORKING-STORAGE SECTION.

       01  CMD-BUFFER.
           05  CMD-TRANSACTION-ID   PIC 9(15) COMP-3.
           05  CMD-ACCOUNT-ID       PIC 9(15) COMP-3.
           05  CMD-AMOUNT           PIC 9(15)V9(4) COMP-3.
           05  CMD-CURRENCY         PIC X(3).
           05  CMD-DIRECTION        PIC X(1).
           05  CMD-IDEMPOTENCY-KEY  PIC X(128).
           05  CMD-CORRELATION-ID   PIC X(128).

       LINKAGE SECTION.
       01  RAW-BUFFER               PIC X(512).

       PROCEDURE DIVISION USING RAW-BUFFER.

       MAIN-LOGIC.
           MOVE RAW-BUFFER TO CMD-BUFFER

           EVALUATE CMD-DIRECTION
              WHEN "D"
                 CALL "POSTTRAN" USING CMD-TRANSACTION-ID
              WHEN "R"
                 CALL "REVTRAN" USING CMD-TRANSACTION-ID
              WHEN "A"
                 CALL "ADJTRAN" USING
                      CMD-ACCOUNT-ID
                      CMD-AMOUNT
                      CMD-CURRENCY
                      "D"
                      CMD-IDEMPOTENCY-KEY
                      CMD-CORRELATION-ID
              WHEN OTHER
                 PERFORM LOG-ERROR
           END-EVALUATE

           GOBACK.

       LOG-ERROR.
           EXEC SQL
                INSERT INTO ERROR_QUEUE (
                    SOURCE, CORRELATION_ID, ERROR_CODE, MESSAGE, CREATED_AT
                ) VALUES (
                    'LEDGER-GATEWAY',
                    :CMD-CORRELATION-ID,
                    'INVALID_CMD',
                    'UNKNOWN DIRECTION',
                    CURRENT_TIMESTAMP
                )
           END-EXEC.

            date.ToDateTime(TimeOnly.MinValue),
            currency,
            operating,
            reserve,
            settle,
            inTransit));
    }

public sealed class ConcurrencyInvariantHarness
{
    private readonly ILedgerGateway _ledger;
    private readonly IDbConnection _db;

    public ConcurrencyInvariantHarness(ILedgerGateway ledger, IDbConnection db)
    {
        _ledger = ledger;
        _db = db;
    }

    public async Task RunAsync(int workers, int iterations, CancellationToken ct)
    {
        var tasks = Enumerable.Range(0, workers)
            .Select(w => WorkerAsync(w, iterations, ct));
        await Task.WhenAll(tasks);

        await AssertDoubleEntryAsync(ct);
    }

    private async Task WorkerAsync(int workerId, int iterations, CancellationToken ct)
    {
        var rnd = new Random(workerId);
        for (var i = 0; i < iterations; i++)
        {
            var corr = $"W{workerId}-I{i}";
            var idem = $"IDEMP-{workerId}-{i}";

            var amount = (decimal)(rnd.Next(1, 1000)) / 100m;

            var req = new PostTransactionRequest(
                accountId: 1000 + workerId,
                Amount: amount,
                CurrencyCode: "USD",
                Direction: i % 2 == 0 ? "D" : "C",
                IdempotencyKey: idem);

            var post = await _ledger.InitiateAndPostAsync(req, corr, ct);

            if (i % 10 == 0 && post.Success)
            {
                await _ledger.ReverseAsync(post.TransactionId, idem + "-REV", corr + "-REV", ct);
            }
        }
    }

    private async Task AssertDoubleEntryAsync(CancellationToken ct)
    {
        var cmd = _db.CreateCommand();
        cmd.CommandText = @"
            SELECT COALESCE(SUM(CASE WHEN DIRECTION='D' THEN AMOUNT ELSE 0 END),0) AS DEBITS,
                   COALESCE(SUM(CASE WHEN DIRECTION='C' THEN AMOUNT ELSE 0 END),0) AS CREDITS
              FROM JOURNAL_LINES";
        using var reader = await cmd.ExecuteReaderAsync(ct);
        if (await reader.ReadAsync(ct))
        {
            var debits = reader.GetDecimal(0);
            var credits = reader.GetDecimal(1);
            if (debits != credits)
                throw new InvalidOperationException($"Invariant violated: D={debits} C={credits}");
        }
    }

    public async Task RecordAsync(string eventType,
                                  string aggregateType,
                                  long aggregateId,
                                  byte[] payload,
                                  string correlationId,
                                  CancellationToken ct)
    {
        using var tx = _db.BeginTransaction();
        var cmd = _db.CreateCommand();
        cmd.Transaction = tx;
        cmd.CommandText = @"
            INSERT INTO DOMAIN_EVENTS (
                EVENT_TYPE, AGGREGATE_TYPE, AGGREGATE_ID,
                OCCURRED_AT, PAYLOAD_BIN, CORRELATION_ID, STATUS
            ) VALUES (
                @etype, @atype, @aid,
                CURRENT_TIMESTAMP, @payload, @corr, 'RECORDED'
            );
            SELECT IDENTITY_VAL_LOCAL() FROM SYSIBM.SYSDUMMY1";
        cmd.Parameters.Add(NewParam(cmd, "@etype", eventType));
        cmd.Parameters.Add(NewParam(cmd, "@atype", aggregateType));
        cmd.Parameters.Add(NewParam(cmd, "@aid", aggregateId));
        cmd.Parameters.Add(NewParam(cmd, "@payload", payload));
        cmd.Parameters.Add(NewParam(cmd, "@corr", correlationId));

        var eventId = Convert.ToInt64(await cmd.ExecuteScalarAsync(ct));

        var outbox = _db.CreateCommand();
        outbox.Transaction = tx;
        outbox.CommandText = @"
            INSERT INTO EVENT_OUTBOX (EVENT_ID, CREATED_AT)
            VALUES (@eid, CURRENT_TIMESTAMP)";
        outbox.Parameters.Add(NewParam(outbox, "@eid", eventId));
        await outbox.ExecuteNonQueryAsync(ct);

        tx.Commit();
    }

    public async Task DispatchPendingAsync(CancellationToken ct)
    {
        var cmd = _db.CreateCommand();
        cmd.CommandText = @"
            SELECT eo.OUTBOX_ID, de.EVENT_ID, de.EVENT_TYPE, de.PAYLOAD_BIN
              FROM EVENT_OUTBOX eo
              JOIN DOMAIN_EVENTS de ON eo.EVENT_ID = de.EVENT_ID
             WHERE eo.DISPATCHED_AT IS NULL";
        using var reader = await cmd.ExecuteReaderAsync(ct);
        var toDispatch = new List<(long OutboxId, long EventId, string Type, byte[] Payload)>();
        while (await reader.ReadAsync(ct))
        {
            toDispatch.Add((
                reader.GetInt64(0),
                reader.GetInt64(1),
                reader.GetString(2),
                (byte[])reader.GetValue(3)));
        }

        foreach (var e in toDispatch)
        {
            // send to Kafka, MQ, etc. (omitted)
            var upd = _db.CreateCommand();
            upd.CommandText = @"
                UPDATE EVENT_OUTBOX
                   SET DISPATCHED_AT = CURRENT_TIMESTAMP
                 WHERE OUTBOX_ID = @oid";
            upd.Parameters.Add(NewParam(upd, "@oid", e.OutboxId));
            await upd.ExecuteNonQueryAsync(ct);
        }
    }

    private static IDbDataParameter NewParam(IDbCommand cmd, string name, object value)
    {
        var p = cmd.CreateParameter();
        p.ParameterName = name;
